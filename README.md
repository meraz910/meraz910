# ⚡ CODE MODE

```cpp
#include <iostream>
#include <string>

using namespace std;

class Developer
{
private:
    string name;
    string goal;

public:

    Developer(string n, string g)
    {
        name = n;
        goal = g;
    }

    void introduce()
    {
        cout << "Hello, I'm " << name << endl;
        cout << "Goal: " << goal << endl;
    }

    void work()
    {
        while (true)
        {
            learn();
            code();
            build();
            debug();
            improve();
        }
    }

    void learn()
    {
        cout << "Learning..." << endl;
    }

    void code()
    {
        cout << "Coding..." << endl;
    }

    void build()
    {
        cout << "Building..." << endl;
    }

    void debug()
    {
        cout << "Debugging..." << endl;
    }

    void improve()
    {
        cout << "Improving..." << endl;
    }
};

int main()
{
    Developer miraj(
        "Miraj",
        "Become a great software engineer"
    );

    miraj.introduce();

    return 0;
}
```
